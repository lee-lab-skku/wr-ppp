% =========================================================
% Automatic reporting week
%
% Week convention:
%   Monday -- Sunday
%
% Month assignment:
%   A week belongs to the month containing its Thursday.
%   This is equivalent to assigning the week to the month
%   containing at least four of its seven days.
%
% Example:
%   Mon Aug 31 -- Sun Sep 6
%   Thursday = Sep 3
%   => September, Week 1
% =========================================================

\usepackage{pgfcalendar}

% ---------------------------------------------------------
% Integer registers
% ---------------------------------------------------------
% \ReportDateJulian:
%   Julian day corresponding to the compilation date.
%
% \WeekdayIndex:
%   Monday=0, ..., Sunday=6.
%
% \ReportWeekNumber:
%   Week number within the reporting month.
% ---------------------------------------------------------

\newcount\ReportDateJulian
\newcount\WeekdayIndex
\newcount\ReportWeekNumber


% =========================================================
% STEP 1
% Convert compilation date into a Julian day.
%
% \year, \month, and \day are built-in TeX values.
% They correspond to the same date displayed by \today.
% =========================================================

\pgfcalendardatetojulian
    {\the\year-\the\month-\the\day}
    {\ReportDateJulian}


% =========================================================
% STEP 2
% Determine the weekday.
%
% PGF convention:
%   Monday    = 0
%   Tuesday   = 1
%   Wednesday = 2
%   Thursday  = 3
%   Friday    = 4
%   Saturday  = 5
%   Sunday    = 6
% =========================================================

\pgfcalendarjuliantoweekday
    {\ReportDateJulian}
    {\WeekdayIndex}


% =========================================================
% STEP 3
% Move to Thursday, which determines the reporting month.
% =========================================================

\advance\ReportDateJulian by 3
\advance\ReportDateJulian by -\WeekdayIndex


% =========================================================
% STEP 4
% Convert Thursday back to a calendar date.
% =========================================================

\pgfcalendarjuliantodate
    {\ReportDateJulian}
    {\ReportYear}
    {\ReportMonth}
    {\ReportThursdayDay}


% =========================================================
% STEP 5
% Determine the monthly week number.
%
% Thursday:
%
%    1--7  -> Week 1
%    8--14 -> Week 2
%   15--21 -> Week 3
%   22--28 -> Week 4
%   29--31 -> Week 5
%
% Formula:
%
%   floor((Thursday day - 1) / 7) + 1
% =========================================================

\ReportWeekNumber=\ReportThursdayDay
\advance\ReportWeekNumber by -1
\divide\ReportWeekNumber by 7
\advance\ReportWeekNumber by 1

% Add a leading zero to a one-digit number.
% 8 -> 08
% 12 -> 12
\newcommand{\TwoDigits}[1]{%
    \ifnum\numexpr#1\relax<10 0\fi
    \number\numexpr#1\relax
}


% Example:
%   2026-08-W3
\newcommand{\ReportWeekLabel}{%
    \ReportYear-\TwoDigits{\ReportMonth}-W\the\ReportWeekNumber%
}


% =========================================================
% Figure helpers
%
% Use \ReportFigure for one figure.
% Use \ReportFigurePair for two independent figures in one row.
% Missing files are replaced with an informative placeholder.
% =========================================================

% Internal helper: image or fixed-size placeholder.
% Arguments: file path, display width, display height.
\newcommand{\ReportImageOrPlaceholder}[3]{%
    \parbox[c][#3][c]{#2}{%
        \centering
        \IfFileExists{#1}
        {%
            \includegraphics[
                width=\linewidth,
                height=#3,
                keepaspectratio
            ]{#1}%
        }
        {%
            \fbox{%
                \parbox[c][
                    \dimexpr#3-2\fboxsep-2\fboxrule\relax
                ][c]{
                    \dimexpr\linewidth-2\fboxsep-2\fboxrule\relax
                }{%
                    \centering
                    \small
                    \textbf{Figure Placeholder}\\[2mm]
                    Add the image at\\
                    \path{#1}%
                }%
            }%
        }%
    }%
}

% One figure.
% Arguments: file path, caption, unique label.
\newcommand{\ReportFigure}[3]{%
    \begin{figure}[H]
        \centering
        \ReportImageOrPlaceholder{#1}{0.72\linewidth}{35mm}
        \caption{#2}
        \label{#3}
    \end{figure}
}

% Two independent figures placed side by side.
% Each figure has its own caption, number, and label.
% Arguments: left path/caption/label, then right path/caption/label.
\newcommand{\ReportFigurePair}[6]{%
    \begin{figure}[H]
        \centering
        \begin{minipage}[t]{0.48\linewidth}
            \centering
            \ReportImageOrPlaceholder{#1}{\linewidth}{35mm}
            \caption{#2}
            \label{#3}
        \end{minipage}%
        \hfill
        \begin{minipage}[t]{0.48\linewidth}
            \centering
            \ReportImageOrPlaceholder{#4}{\linewidth}{35mm}
            \caption{#5}
            \label{#6}
        \end{minipage}
    \end{figure}
}
